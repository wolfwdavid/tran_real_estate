3. `apps/web/src/lib/base-path.js` — the validator from the RESEARCH `svelte.config.js`, extracted so it can be unit-tested (RESEARCH Test Map: "extract the validator and test it"). JSDoc-typed because `checkJs` is on:
```js
const VALID_BASE = /^\/[A-Za-z0-9._~-]+(\/[A-Za-z0-9._~-]+)*$/;

/**
 * Resolve and validate SvelteKit's `paths.base` from the environment.
 * @param {Record<string, string | undefined>} [env]
 * @returns {string}
 */
export function resolveBasePath(env = process.env) {
	const base = env.BASE_PATH ?? '';
	if (base !== '' && !VALID_BASE.test(base)) {
		throw new Error(
			`BASE_PATH must be '' or a root-relative path like '/tran_real_estate' (got ${JSON.stringify(base)}). ` +
				`On Git Bash set MSYS_NO_PATHCONV=1 or use scripts/build-web.mjs.`
		);
	}
	return base;
}

/**
 * @param {Record<string, string | undefined>} [env]
 * @returns {string}
 */
export function resolveBuildDir(env = process.env) {
	return env.BUILD_DIR ?? 'build';
}
```

const execPath = process.env.npm_execpath;
const [bin, pre] = execPath ? [process.execPath, [execPath]] : ['corepack', ['pnpm']];
const probe = spawnSync(bin, [...pre, '--version'], { encoding: 'utf8' });
const version = (probe.stdout ?? '').trim();
if (version !== '11.27.0') console.warn(`build-web: pnpm ${version || 'unknown'} found; pinned version is 11.27.0`);
