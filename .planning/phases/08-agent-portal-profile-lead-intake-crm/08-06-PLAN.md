```ts
export type FillableField = (typeof FILLABLE_FIELDS)[number];
export interface TargetRef { id: string; name: string }
export type LeadDiffRow =
	| { kind: 'new'; lead: IncomingLead }
	| { kind: 'update'; lead: IncomingLead; target: TargetRef; fills: FillableField[]; keeps: { field: FillableField; existing: string; incoming: string }[]; logsInquiry: boolean }
	| { kind: 'duplicate'; lead: IncomingLead; target: TargetRef }
	| { kind: 'conflict'; lead: IncomingLead; emailMatch: TargetRef; phoneMatch: TargetRef }
	| { kind: 'same-as-row'; lead: IncomingLead; firstRow: number };
export interface LeadDiff { rows: LeadDiffRow[]; counts: { new: number; update: number; duplicate: number; conflict: number; sameAsRow: number } }
export function diffLeads(incoming: readonly IncomingLead[], existing: readonly Contact[]): LeadDiff;
```
