```ts
export function bytesToBase64(bytes: Uint8Array): string; // chunked String.fromCharCode + btoa (0x8000 chunks)
export function base64ToBytes(b64: string): Uint8Array;
export type LogoResult = { ok: true; logo: StoredLogo } | { ok: false; reason: 'too-large'; size: number } | { ok: false; reason: 'not-png-or-svg' } | { ok: false; reason: 'unreadable' };
export async function prepareLogo(file: File, deps?: { rasterize?: Rasterize }): Promise<LogoResult>;
export type Rasterize = (bytes: Uint8Array, mime: 'image/png' | 'image/svg+xml', target: { width: number; height: number }) => Promise<Uint8Array>; // default: Blob → object URL → <img>.decode() → canvas drawImage → canvas.toBlob('image/png') → bytes; revoke the URL
```
   Steps: `checkLogoFile({ size: file.size, bytes: first 1024 bytes })` before reading the whole file; PNG with longest side ≤ 1024 → store original bytes; PNG larger → rasterize to rasterTargetSize; SVG → size from svgIntrinsicSize (else 1024×512, contain-fit) → rasterize to PNG; if the produced PNG > 524288 bytes, retry at 75% dimensions up to 3 times, else `too-large`. Result `{ base64, mime: 'image/png', width, height, bytes: length }`. Never store SVG.
2. logo-url.ts: `export function logoObjectUrl(logo: StoredLogo): string` (Blob from base64ToBytes, type image/png) — callers revoke in effect cleanup.
