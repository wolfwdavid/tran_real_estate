```ts
export interface ImportPlan { adds: IncomingLead[]; updates: { contactId: string; fills: Partial<Record<FillableField, string>>; inquiryMessage?: string }[]; skipped: number }
export function planFromDiff(rows: readonly LeadDiffRow[], include: ReadonlySet<number /* rowNumber */>, edits?: ReadonlyMap<number, Partial<Pick<IncomingLead, 'name' | 'email' | 'phone'>>>): ImportPlan;
// 'new' rows in include → adds (with edits applied: free-form rows are editable in the preview);
// 'update' rows in include → updates with fills taken from lead fields named in row.fills, inquiryMessage = lead.message when row.logsInquiry;
// everything else (duplicate, conflict, same-as-row, unchecked) → skipped count
export async function commitImport(plan: ImportPlan, now?: string): Promise<{ added: number; updated: number; skipped: number }>;
// ONE db.transaction('rw', contacts, activities) for the whole plan:
//  adds → buildContactRecord + 'added' activity (ADDED_TEXT[lead.source]) + when lead.message: 'inquiry' activity `Inquiry received: "${first80}${len > 80 ? '…' : ''}"`
//  updates → re-read target inside the tx; for each fill only write when the stored field is still blank (never overwrite); re-normalize email/phone; 'Details updated' activity when any field filled; 'inquiry' activity when inquiryMessage
// any throw → whole transaction aborts (Dexie) → nothing written; rethrow ImportCommitError for the UI's "Import didn't finish" copy
export class ImportCommitError extends Error {}
```
