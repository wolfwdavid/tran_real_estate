export interface KeyPort { get(): string | null; state(): RentCastKeyState; onChange(cb: () => void): () => void }

export interface CapPrompt { used: number; cap: number; requestKey: string }
export interface DataRuntime {
	readonly mode: { source: 'sample' | 'rentcast'; reason: 'no-key' | 'locked' | 'chosen' | 'live' } | null; // null until start() resolves
	readonly usage: (UsageSnapshot & { state: MeterState }) | null;
	readonly gateway: DataGateway | null;
	readonly capPrompt: CapPrompt | null;
	readonly lastRemoteChange: { type: DataMessage['type']; seq: number } | null; // seq increments on every message from another tab
	start(): Promise<() => void>;                       // call from onMount; resolves to a cleanup function
	requestSpend<T>(req: DataRequest, kind?: 'get' | 'refresh'): Promise<GatewayResult<T>>;
	confirmCapPrompt(): Promise<void>;
	cancelCapPrompt(): void;
	setSourcePreference(p: DataSourcePref): Promise<void>;
	saveCap(cap: number): Promise<void>;
	testKey(): Promise<KeyTestRecord>;
