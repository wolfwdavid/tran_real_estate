// grants.ts
export interface OverCapGrant { readonly requestKey: string; readonly consumed: boolean }
export function issueGrant(requestKey: string): OverCapGrant;      // registers in a module-private WeakSet
export function consumeGrant(grant: OverCapGrant | undefined, requestKey: string): boolean; // true once, only for issued grants with a matching key
// ledger-repo.ts
export interface SpendRequest { endpoint: EndpointId; key: string; intent: LedgerIntent }
export interface UsageSnapshot extends UsageSummary { cap: number; periodKey: string; resetsAt: number }
export interface LedgerRepo {
	reserve(req: SpendRequest, grant?: OverCapGrant): Promise<string>; // ledger row id; throws GatewayError BudgetCapReached { used, cap, resetsAt, requestKey }
	settle(id: string, status: number): Promise<void>;
	record(req: SpendRequest, status: number): Promise<string>;       // unreserved, already settled (key tests)
	usage(): Promise<UsageSnapshot>;
	readCap(): Promise<number>;
	writeCap(cap: number): Promise<void>;                             // RangeError outside 1..1000 or non-integer
}
export function createLedgerRepo(db: DataDb, clock: () => number): LedgerRepo;
// cache-repo.ts
export interface CacheRepo { get(key: string): Promise<ApiCacheRow | undefined>; put(row: ApiCacheRow): Promise<void>; findListingById(id: string): Promise<{ row: ApiCacheRow; item: unknown } | undefined>; prune(now: number, maxAgeMs?: number): Promise<number> }
export function createCacheRepo(db: DataDb): CacheRepo;
