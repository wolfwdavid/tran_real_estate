```ts
// location
export type NormalizedLocation = { kind: 'zip'; zipCode: string; label: string } | { kind: 'city'; city: string; state: string; label: string };
export type LocationResult = ({ ok: true } & NormalizedLocation) | { ok: false; error: 'empty' | 'invalid-zip' | 'missing-state' | 'unknown-state' };
export function normalizeLocation(input: string): LocationResult;
export const parseLocation = normalizeLocation;
export const US_STATES: readonly { abbr: string; name: string }[]; // 50 states + DC = 51 entries
// listings
export const PROPERTY_TYPES = ['Single Family','Condo','Townhouse','Manufactured','Multi-Family','Apartment','Land'] as const;
export type PropertyType = (typeof PROPERTY_TYPES)[number] | 'Other';
export interface ListingFilters { minPrice?: number; maxPrice?: number; minBedrooms?: number; maxBedrooms?: number; minBathrooms?: number; propertyTypes?: readonly PropertyType[]; listedWithinDays?: number }
export function filterListings<T extends Pick<Listing, 'price' | 'bedrooms' | 'bathrooms' | 'propertyType' | 'listedDate'>>(rows: readonly T[], filters: ListingFilters, now: number): T[];
export const LISTINGS_LIMIT = 500;
export function buildListingsRequest(location: NormalizedLocation, endpoint: ListingsEndpoint, narrow?: ListingFilters): ListingsRequest;
export function buildListingByIdRequest(id: string): ListingByIdRequest;
export function toRentCastFilterParams(filters: ListingFilters): Record<string, string | readonly string[]>;
export const isTruncated: (rowCount: number, limit?: number) => boolean; // rowCount === limit (Pitfall 7: X-Total-Count unreadable)
export function shouldOfferNarrow(truncated: boolean, filters: ListingFilters): boolean;
```
