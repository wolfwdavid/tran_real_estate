export interface SnapshotItemLike { price: number | null; status: string | null; removedDate: string | null; firstSeenAt: number }
export interface SnapshotLike { takenAt: number; truncated: boolean; items: Record<string, SnapshotItemLike> }
export interface ObservedItem { id: string; price: number | null; status: string | null; removedDate: string | null }
export interface SnapshotDiff { new: string[]; priceChanged: { id: string; from: number | null; to: number | null }[]; gone: string[]; unknown: string[] }
export function buildSnapshot(prev: SnapshotLike | null, observed: readonly ObservedItem[], opts: { takenAt: number; truncated: boolean }): SnapshotLike;
export function diffSnapshots(prev: SnapshotLike | null, next: SnapshotLike): SnapshotDiff;
export function countNewSince(snapshot: SnapshotLike | null, lastSeenAt: number | null): number;
export const DUE_AFTER_MS = 86_400_000; export const CLAIM_TTL_MS = 120_000; export const CONFIRM_ABOVE = 3;
export interface DueCandidate { id: string; lastRunAt: number | null; claimedAt: number | null; requestKey: string; cached: boolean }
export function isDue(s: Pick<DueCandidate, 'lastRunAt' | 'claimedAt'>, now: number): boolean;
export function dueSearches<T extends DueCandidate>(searches: readonly T[], now: number): T[];
export interface RunPlan<T> { run: T[]; wait: T[]; calls: number; needsConfirm: boolean; blockedByCap: boolean }
export function planRun<T extends DueCandidate>(due: readonly T[], opts: { used: number; cap: number; mode: 'live' | 'sample' }): RunPlan<T>;
