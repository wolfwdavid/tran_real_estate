```ts
export const PLAN_LIMIT = 50; export const DEFAULT_CAP = 40; export const CAP_MIN = 1; export const CAP_MAX = 1000; export const OVERAGE_USD = 0.2;
export const METER_THRESHOLDS = { nearMaxLeft: 5 } as const;
export function validateCap(input: string): CapValidation { const s = input.trim(); if (s === '') return { ok: false, error: 'empty' }; if (!/^-?\d+$/.test(s)) return { ok: false, error: 'not-integer' }; const n = Number(s); if (n < CAP_MIN) return { ok: false, error: 'below-min' }; if (n > CAP_MAX) return { ok: false, error: 'above-max' }; return { ok: true, value: n, aboveLimit: n > PLAN_LIMIT }; }
export const matchesOverageWord = (input: string) => input.trim().toLowerCase() === 'overage';
export const canSpend = (used: number, cap: number, cost = 1) => used + cost <= cap;
export function meterState(used: number, cap: number, planLimit = PLAN_LIMIT): MeterState { if (used >= planLimit) return 'past'; if (used >= cap) return 'at'; if (cap - used <= METER_THRESHOLDS.nearMaxLeft) return 'near'; return 'ok'; }
const RANK = { ok: 0, near: 1, at: 2, past: 3 } as const;
export function meterTransition(prev: MeterState | null, next: MeterState) { if (prev === null || RANK[next] <= RANK[prev]) return null; return next === 'ok' ? null : next; }
export const confirmVariant = (used: number, planLimit = PLAN_LIMIT) => (used + 1 > planLimit ? 'past-plan' : 'within-plan');
export const overageCostUsd = (cap: number) => (Math.max(cap - PLAN_LIMIT, 0) * Math.round(OVERAGE_USD * 100)) / 100;
```
