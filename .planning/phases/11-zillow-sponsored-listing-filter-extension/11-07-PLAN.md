  release:
    name: build, verify and publish the extension zip
    if: vars.EXTENSION_RELEASES_ENABLED == 'true'
    runs-on: ubuntu-24.04
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - uses: pnpm/action-setup@ea17c68df8912ef543352723c149a84f56e3d413 # v6.1.0
      # No cache on a release job: a cache written by another ref could poison the published zip.
      - uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0
        with:
          node-version-file: .nvmrc
      - run: pnpm install --frozen-lockfile
      - name: Tag matches the extension package version
        run: test "$GITHUB_REF_NAME" = "ext-v$(node -p "require('./apps/extension/package.json').version")"
      - run: pnpm vitest run --project extension

      - run: node scripts/assert-extension.mjs apps/extension/.output/chrome-mv3 --zip-dir apps/extension/.output
      - name: Checksums
