```ts
import { COPY } from './copy';
import { GET_COUNT, type CountPush, type CountResponse, type GetCountMessage } from './messages';
export type PopupState =
	| { kind: 'loading' } | { kind: 'counting'; count: number } | { kind: 'zero' } | { kind: 'off' }
	| { kind: 'not-zillow' } | { kind: 'not-running' } | { kind: 'storage-error' };
export interface ActiveTab { id?: number; url?: string }
export interface PopupDeps {
	readEnabled(): Promise<boolean>;
	writeEnabled(value: boolean): Promise<void>;
	activeTab(): Promise<ActiveTab | undefined>;
	sendToTab(tabId: number, message: GetCountMessage): Promise<CountResponse>;
	onCountPush(listener: (push: CountPush, senderTabId: number | undefined) => void): () => void;
}
export interface PopupStatus { enabled: boolean | null; tabId: number | undefined; state: PopupState }
export function isZillowUrl(url: string | undefined): boolean; // URL parse; protocol http/https; hostname === 'zillow.com' || endsWith('.zillow.com')
export async function loadStatus(deps: PopupDeps): Promise<PopupStatus>;
export function applyPush(state: PopupState, push: CountPush, activeTabId: number | undefined, senderTabId: number | undefined): PopupState;
export function countLabel(n: number): string;
export function stateCopy(state: PopupState): string;
```
