- `src/lib/content-wiring.ts` (all logic, DI'd so it is unit-testable):
```ts
import { startFilter, type FilterController } from './filter';
import { createResponder, type CountPush } from './messages';
import type { EnabledSetting } from './settings';
export interface RuntimeLike { sendMessage(m: CountPush): Promise<unknown>; onMessage: { addListener(l: (...a: never[]) => unknown): void; removeListener(l: (...a: never[]) => unknown): void } }
export interface CtxLike { setTimeout(cb: () => void, ms: number): number; addEventListener(target: Window, type: 'wxt:locationchange', cb: () => void): void; onInvalidated(cb: () => void): void }
export async function wireContent(ctx: CtxLike, deps: { doc: Document; win: Window; runtime: RuntimeLike; setting: Pick<EnabledSetting, 'getValue' | 'watch'> }): Promise<FilterController> {
	const filter = await startFilter({
		doc: deps.doc,
		setting: deps.setting,
		timers: { setTimeout: (cb, ms) => ctx.setTimeout(cb, ms), clearTimeout: (id) => deps.win.clearTimeout(id) },
		onNavigate: (cb) => ctx.addEventListener(deps.win, 'wxt:locationchange', () => cb()),
		publish: (push) => { deps.runtime.sendMessage(push).catch(() => {}); }
	});
	const responder = createResponder(() => ({ enabled: filter.enabled(), count: filter.count() }));
	deps.runtime.onMessage.addListener(responder);
	ctx.onInvalidated(() => { deps.runtime.onMessage.removeListener(responder); filter.stop(); });
	return filter;
}
```
Adjust the listener types to whatever `wxt/browser`'s `runtime.onMessage` accepts so `browser.runtime` satisfies `RuntimeLike` without casts (record if a narrow cast is unavoidable).
