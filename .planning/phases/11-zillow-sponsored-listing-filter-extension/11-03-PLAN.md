```ts
import { type Category, type FilterConfig, type OptInCategory, MAX_ATTR_LEN, MAX_TEXT_LEN, PILL_TAG, normalizeLabel } from './config';
export type HitCategory = Category | OptInCategory;
export interface Hit { category: HitCategory; via: 'ad-slot' | 'label' | 'attribute' }
export interface ClassifyOptions { optIn?: readonly OptInCategory[] }
export function findContainer(root: ParentNode, cfg: FilterConfig): Element | null; // first resultsContainers match in order; never body/html
export function itemsIn(container: Element, cfg: FilterConfig): Element[];
export function classify(item: Element, cfg: FilterConfig, opts?: ClassifyOptions): Hit | null;
```
- `itemsIn`: union of `container.querySelectorAll(sel)` for each itemRoots and adSlots selector; if empty, use cardRoots matches; dedupe with a Set; drop any element that has an ancestor also in the set (outermost wins = the dim target per RESEARCH "Dim target").
- `classify` order: (1) if `item.matches(adSel) || item.querySelector(adSel)` for any adSlots selector → `{ category: 'ad', via: 'ad-slot' }`. (2) Build a `Map<string, HitCategory>` from cfg.labels (plus cfg.optInLabels entries only for categories in opts.optIn). (3) Walk text nodes with `item.ownerDocument.createTreeWalker(item, NodeFilter.SHOW_TEXT)`, skipping any node inside `PILL_TAG`, `script`, `style`; for trimmed text with `0 < length <= MAX_TEXT_LEN`, look up `normalizeLabel(text)` → hit via 'label'.
