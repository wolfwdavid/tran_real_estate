```ts
// onboarding-state.ts — localStorage only (no Dexie, no $lib/storage); read in onMount, never at import (prerender has no localStorage).
export const ONBOARDING_KEY = 'tran.onboarding.v1';
export type OnboardingPortal = 'renter' | 'investor' | 'agent';
export interface OnboardingState { status: 'new' | 'touring' | 'done' | 'hidden'; portal?: OnboardingPortal; step?: 0 | 1 | 2 | 3; lastPortal?: OnboardingPortal }
type KV = Pick<Storage, 'getItem' | 'setItem'>;
export function parseOnboarding(raw: string | null): OnboardingState { /* strict whitelist per behavior */ }
const defaultStorage = (): KV | undefined => { try { return globalThis.localStorage; } catch { return undefined; } };
export function readOnboarding(storage: KV | undefined = defaultStorage()): OnboardingState { try { return parseOnboarding(storage?.getItem(ONBOARDING_KEY) ?? null); } catch { return { status: 'new' }; } }
export function writeOnboarding(state: OnboardingState, storage: KV | undefined = defaultStorage()): boolean { try { storage?.setItem(ONBOARDING_KEY, JSON.stringify(state)); return Boolean(storage); } catch { return false; } }
```
