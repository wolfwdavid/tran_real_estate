```ts
// payload.ts
export const SCREEN_CLASSES = ['small', 'medium', 'large'] as const;
export type ScreenClass = (typeof SCREEN_CLASSES)[number];
export type AnalyticsPortal = 'renter' | 'investor' | 'agent' | 'none';
export interface BeaconBody { type: 'event'; payload: { website: string; hostname: string; url: string; referrer?: string; screen: ScreenClass; data: { portal: AnalyticsPortal } } }
export function screenClass(width: number): ScreenClass { if (!Number.isFinite(width) || width < 640) return 'small'; return width < 1024 ? 'medium' : 'large'; }
export function stripBase(pathname: string, base: string): string { /* cut at first ? or #; if base && (p === base || p.startsWith(base + '/')) remove base; return p || '/' with a leading '/' */ }
export function portalOf(path: string): AnalyticsPortal { const seg = path.split('/')[1]; return seg === 'renter' || seg === 'investor' || seg === 'agent' ? seg : 'none'; }
/** Full origin URL, never a bare host (Pitfall 7); undefined when same-site, non-http(s) or unparseable. */
export function referrerOrigin(ref: string, selfHost: string): string | undefined { try { const u = new URL(ref); return u.hostname === selfHost || !/^https?:$/.test(u.protocol) ? undefined : `${u.protocol}//${u.hostname}/`; } catch { return undefined; } }
export function buildPageview(loc: { pathname: string; hostname: string }, referrer: string, width: number, websiteId: string, base = ''): BeaconBody { const url = stripBase(loc.pathname, base); const ref = referrerOrigin(referrer, loc.hostname); return { type: 'event', payload: { website: websiteId, hostname: loc.hostname, url, ...(ref ? { referrer: ref } : {}), screen: screenClass(width), data: { portal: portalOf(url) } } }; }
```

```ts
// beacon.ts  (the ONLY fetch in lib/analytics; the folder is fetch-allowlisted)
import { buildPageview, type BeaconBody } from './payload';
import { trackingAllowed } from './consent';
export function send(body: BeaconBody, cfg: { host: string }, fetchImpl?: typeof fetch): void {
	if (!cfg.host) return; // no-op build
	try {
		const f = fetchImpl ?? globalThis.fetch;
		void f(`${cfg.host}/api/send`, { method: 'POST', body: JSON.stringify(body), mode: 'no-cors', credentials: 'omit', keepalive: true, referrerPolicy: 'no-referrer', cache: 'no-store' }).catch(() => {});
	} catch { /* analytics must never surface errors */ }
}
export interface TrackerConfig { host: string; websiteId: string; base: string; fetchImpl?: typeof fetch; nav?: { globalPrivacyControl?: boolean; doNotTrack?: string | null }; win?: { doNotTrack?: string | null } }
export function createTracker(cfg: TrackerConfig) {
	const enabled = Boolean(cfg.host && cfg.websiteId);
	return {
		pageview(url: URL, referrer: string, width: number): void {
			if (!enabled) return;
			const nav = cfg.nav ?? (globalThis.navigator as TrackerConfig['nav']) ?? {};
			const win = cfg.win ?? (globalThis as TrackerConfig['win'] & object);
			if (!trackingAllowed(nav, win)) return;
			send(buildPageview({ pathname: url.pathname, hostname: url.hostname }, referrer, width, cfg.websiteId, cfg.base), cfg, cfg.fetchImpl);
		}
	};
}
```
