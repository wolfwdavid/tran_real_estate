```ts
import { monthlyPayment } from './amortization';
const ratio = (num: number, den: number): number | null => (den === 0 ? null : num / den);
const pctToBps = (p: number) => Math.round(p * 100);
const egi = (monthlyIncome: Cents, vacancyPct: number): Cents => { const gross = monthlyIncome * 12; return gross - mulBps(gross, pctToBps(vacancyPct)); };
export interface CapRateInput { price: Cents; monthlyRent: Cents; otherMonthlyIncome: Cents; vacancyPct: number; monthlyOpex: Cents }
export interface CapRateResult { capRate: number | null; noiYearly: Cents; egiYearly: Cents; opexYearly: Cents }
export function capRateCalc(i: CapRateInput): CapRateResult { const egiYearly = egi(i.monthlyRent + i.otherMonthlyIncome, i.vacancyPct); const opexYearly = i.monthlyOpex * 12; const noiYearly = egiYearly - opexYearly; return { capRate: ratio(noiYearly, i.price), noiYearly, egiYearly, opexYearly }; }
export interface CashOnCashInput { price: Cents; downPaymentPct: number; closingCosts: Cents; interestRatePct: number; termYears: number; monthlyRent: Cents; vacancyPct: number; monthlyOpex: Cents }
export interface CashOnCashResult { cashOnCash: number | null; cashFlowYearly: Cents; cashFlowMonthly: Cents; totalCashInvested: Cents; monthlyPayment: Cents; noiYearly: Cents; loanAmount: Cents }
export function cashOnCashCalc(i: CashOnCashInput): CashOnCashResult { /* down = mulBps(price, pctToBps(downPaymentPct)); loan = price - down; pay = monthlyPayment(loan, rate, term); noi via capRateCalc pieces with other income 0; cfY = noi - pay*12; cfM = Math.round(cfY / 12) (half away from zero, matching Phase 7 rounding helper if exported); invested = down + closingCosts; cashOnCash = ratio(cfY, invested) */ }
export interface DscrInput { monthlyRent: Cents; vacancyPct: number; monthlyOpex: Cents; loanAmount: Cents; interestRatePct: number; termYears: number }
export interface DscrResult { applicable: boolean; dscr: number | null; noiYearly: Cents; debtServiceYearly: Cents; monthlyPayment: Cents; cashAfterDebtYearly: Cents }
export function dscrCalc(i: DscrInput): DscrResult { /* applicable = loanAmount > 0; pay = applicable ? monthlyPayment(...) : 0; debt = pay*12; dscr = applicable ? ratio(noi, debt) : null */ }
```

```ts
export type FieldError = { ok: false; code: 'required' | 'nan' | 'min' | 'max' | 'price-zero'; min?: number; max?: number };
```
