```ts
export type IncomePeriod = 'year' | 'month';
export type MoneyParse = number | null | { error: 'invalid' | 'negative' };
export function parseMoneyInput(s: string): MoneyParse     // strips '$', ',', spaces; '' -> null
export const round25 = (x: number) => Math.round(x / 25) * 25;
export const floor25 = (x: number) => Math.floor(x / 25) * 25;
export interface AffordabilityInput { income: number | null; period: IncomePeriod; debts: number; savings: number }
export interface AffordabilityResult { monthlyIncome: number; pct25: number; pct30: number; debtCheck: number; low: number; high: number; onlyTop: boolean; debtsNote: boolean; savingsCoverRentUpTo: number }
export function affordability(i: AffordabilityInput): AffordabilityResult | null
export function rentFeedback(rent: number, monthlyIncome: number, debts: number): { kind: 'within' | 'above30' | 'aboveDebt'; rentPct: number; totalPct: number }
export const landlordIncome = (rent: number) => rent * 40;
export const moveInCash = (rent: number) => rent * 2;
export function sliderMax(top: number): number             // Math.ceil(Math.max(10000, 2 * top) / 500) * 500
```
   Math exactly per UI-SPEC: `M = period === 'year' ? income / 12 : income`; `pct25 = 0.25*M`, `pct30 = 0.30*M`, `debtCheck = max(0, 0.40*M - debts)`; `low = round25(pct25)`; `high = round25(min(pct30, debtCheck))`; `onlyTop = high < low`; `debtsNote = debtCheck < pct25`; `savingsCoverRentUpTo = floor25(savings / 2)`. `rentFeedback`: `rentPct = Math.round(rent / M * 100)`, `totalPct = Math.round((rent + debts) / M * 100)`; compare with the unrounded ratios: `(rent + debts) / M > 0.40` → `'aboveDebt'`, else `rent / M > 0.30` → `'above30'`, else `'within'`. Return `null` from `affordability` when income is null, NaN, or ≤ 0. Header comment: "General guideline math only. No persistence, no I/O (Phase 10 reuses this on a marketing page)."
