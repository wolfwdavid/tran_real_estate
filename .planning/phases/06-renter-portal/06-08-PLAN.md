```ts
export const SHORTLIST_STATUSES = ['saved', 'contacted', 'toured', 'applied', 'signed'] as const;
export type ShortlistStatus = (typeof SHORTLIST_STATUSES)[number];
export const STATUS_LABELS: Record<ShortlistStatus, string> = { saved: 'Saved', contacted: 'Contacted', toured: 'Toured', applied: 'Applied', signed: 'Signed' };
export const MAX_NOTES = 2000; export const NOTES_PREVIEW = 120; export const MAX_SAVED_SEARCHES = 10;
export const RENTER_SORT_IDS = ['newest', 'rent-asc', 'rent-desc', 'beds', 'size', 'dom'] as const;
export const shortlistItemSchema = /* Syncable shape extended with */ { listingId: z.string() (min 1), listing: z.looseObject({ id: z.string(), price: z.nullable(z.number()) }), provenance: z.object({ source: z.enum(['rentcast','sample']), fetchedAt: z.number() }), status: z.enum(SHORTLIST_STATUSES), statusChangedAt: z.number(), statusHistory: z.array(z.object({ status: z.enum(SHORTLIST_STATUSES), at: z.number() })), notes: z.string().check(z.maxLength(MAX_NOTES)) };
export type ShortlistItem = z.infer<typeof shortlistItemSchema>;
export const renterSearchMetaSchema = /* Syncable + */ { location: z.union([zip (/^\d{5}$/), city {city: min 1, state: /^[A-Z]{2}$/}]), filters: { priceMin, priceMax: nullable int >= 0; beds: union of 'any',0,1,2,3,'4+'; types: array(enum RENTER_PROPERTY_TYPES); listedWithin: union 'any',7,14,30,60 }, sort: z.enum(RENTER_SORT_IDS) };
export type RenterSearchMeta = z.infer<typeof renterSearchMetaSchema>;
export function changeStatus(item: ShortlistItem, status: ShortlistStatus, now: number): ShortlistItem
export function needsRemoveConfirm(item: Pick<ShortlistItem, 'status' | 'notes'>): boolean
export function shortlistSummary(items: readonly ShortlistItem[]): { total: number; contacted: number; applied: number }
export function groupByStatus(items: readonly ShortlistItem[]): Record<ShortlistStatus, ShortlistItem[]>
export function notesPreview(notes: string): string
export const canSaveAnother = (count: number) => count < MAX_SAVED_SEARCHES;
```
