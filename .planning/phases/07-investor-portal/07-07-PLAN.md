```ts
// compare.ts
export type CompareRowKey = 'price' | 'rent' | 'cashNeeded' | 'cashFlow' | 'coc' | 'capRate' | 'noiAnnual' | 'dscr' | 'grm' | 'rentToPrice';
export interface CompareRowDef { key: CompareRowKey; label: string; direction: 'max' | 'min' | null }
export const COMPARE_ROWS: readonly CompareRowDef[];
// in order: Purchase price (null) · Monthly rent (null) · Total cash needed (min) · Monthly cash flow (max) · Cash-on-cash return (max) · Cap rate (max) · Net operating income (annual) (max) · DSCR (max) · GRM (min) · 1% rule (rent-to-price) (max)
export const MIN_COMPARE = 2; export const MAX_COMPARE = 3;
export function bestInRow(values: readonly (number | null)[], direction: 'max' | 'min'): number[];
export interface CompareRow extends CompareRowDef { values: (number | null)[]; display: string[]; marked: number[]; marker: 'Highest' | 'Lowest' | null }
export function buildComparison(deals: readonly { inputs: DealInputs; analysis: DealAnalysis }[]): CompareRow[]; // throws RangeError outside 2..3
// quick-metrics.ts
export interface QuickMetricsListing { priceCents: number; bedrooms: number | null }
export interface QuickMetrics { rentCents: number; rentToPricePct: number | null; grm: number | null; estCapRatePct: number | null; basis: 'bedrooms' | 'all'; month: string | null; listings: number | null; expenseBps: number }
export function quickMetrics(listing: QuickMetricsListing, stats: MarketStatsInput | null, expenseBps?: number): QuickMetrics | null;
export interface BuyBoxTargets { minCapRatePct: number | null; minRentToPricePct: number | null }
export function meetsTargets(q: QuickMetrics | null, t: BuyBoxTargets): boolean | null; // true if no targets set; null if q is null and a target is set
export function quickBasisLine(q: QuickMetrics, beds: number | null): string;
```
