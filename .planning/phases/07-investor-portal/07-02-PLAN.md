```ts
// packages/core/src/numbers/extract.ts
export interface NumeralToken {
	start: number; end: number;   // offsets into the ORIGINAL text (pre-strip replaces with spaces so indices survive)
	raw: string;                  // exact matched substring
	value: number;                // sign × number × scale
	decimals: number;             // digits after the decimal point as typed
	scale: number;                // 1, 1e3 (k|K|thousand), 1e6 (M|million)
	sig: number;                  // significant digits as typed (leading zeros dropped; integers without decimals also drop trailing zeros: "15,000" -> 2, "1,064.48" -> 6, "0.2" -> 1, "7.5" -> 2, "20" -> 1)
	unit: '%' | 'x' | null;
	currency: boolean;
}
export function extractNumerals(text: string): NumeralToken[];
// packages/core/src/numbers/check.ts
export const NUMERIC_CATEGORY = 'numeric-provenance';
export interface AllowedNumber { value: number; source?: string }
export interface NumberFinding {
	start: number; end: number; text: string;
	severity: 'block' | 'warn';
	category: string;                       // default NUMERIC_CATEGORY; Phase 9 passes its own
	reason: 'not-in-allowed' | 'spelled-out';
}
export function matchesAllowed(tok: NumeralToken, allowed: number): boolean;
export function checkNumbers(text: string, allowed: readonly (number | AllowedNumber)[], opts?: { category?: string }): NumberFinding[]; // sorted by start; includes spelled-out warns
export function allowedFromText(text: string, source?: string): AllowedNumber[]; // = extractNumerals(text) values
// packages/core/src/numbers/spelled.ts
export function findSpelledQuantities(text: string, category?: string): NumberFinding[];
```
Matching rule (RESEARCH Pattern 5, final rule, prototyped 2026-10-04):
`halfUnit = 0.5 × 10^(−decimals) × scale`;
`match(n, a) := sign(n.value) == sign(a) && ( |n.value − a| ≤ halfUnit + 1e-9 || (n.sig ≥ 2 && |n.value − a| ≤ 0.005 × |a|) )`.
Zero has sign 0 and matches only zero-signed allowed values. Matching is unit-agnostic by design (documented).
