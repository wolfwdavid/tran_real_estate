```ts
// types.ts
export interface BedroomStat { bedrooms: number; medianRent?: number | null; medianPrice?: number | null; totalListings?: number | null }
export interface RentalMonth { medianRent?: number | null; totalListings?: number | null; dataByBedrooms?: BedroomStat[] }
export interface SaleMonth { medianPrice?: number | null; totalListings?: number | null }
export interface MarketStatsInput {
	zipCode: string;
	rentalData?: { medianRent?: number | null; totalListings?: number | null; lastUpdatedDate?: string | null; dataByBedrooms?: BedroomStat[]; history?: Record<string, RentalMonth> } | null;
	saleData?: { medianPrice?: number | null; totalListings?: number | null; lastUpdatedDate?: string | null; history?: Record<string, SaleMonth> } | null;
}
export type SeriesKey = 'studio' | '1' | '2' | '3' | '4plus';
export const SERIES_LABELS: Record<SeriesKey, string>; // Studio · 1 bedroom · 2 bedrooms · 3 bedrooms · 4+ bedrooms
export interface RentSeries { key: SeriesKey; label: string; points: (number | null)[]; counts: (number | null)[] } // aligned to months[]
export interface RentTrend { months: string[]; series: RentSeries[] }   // months continuous 'YYYY-MM' ascending
export interface PriceToRent { months: string[]; points: (number | null)[] }
// series.ts
export function monthRange(keys: string[]): string[];                 // continuous inclusive range from min to max key
export function rentTrendSeries(stats: MarketStatsInput): RentTrend;
export function priceToRentSeries(stats: MarketStatsInput): PriceToRent;
export function hasEnoughData(points: (number | null)[][]): boolean;  // true iff some series has ≥ 2 non-null points
export function currentStats(stats: MarketStatsInput): { medianRent: number | null; medianPrice: number | null; priceToRent: number | null; listings: number | null; lastUpdated: string | null };
export function zipMedianRentForBeds(stats: MarketStatsInput, beds: number | null): { rentDollars: number; basis: 'bedrooms' | 'all'; month: string | null; listings: number | null } | null;
// summary.ts
export function timeRangeLine(months: string[]): string;              // "Monthly medians, Oct 2024 to Sep 2026"
export function rentTrendSummary(zip: string, t: RentTrend, opts?: { sample?: boolean }): string;
export function priceToRentSummary(zip: string, p: PriceToRent, opts?: { sample?: boolean }): string;
export function fewListingsSuffix(count: number | null): string;      // "" or " (few listings: 3)" when count < 5
// watched-zips.ts
export const MAX_WATCHED_ZIPS = 5;
export function validateZip(input: string): { ok: true; zip: string } | { ok: false; error: 'invalid' };
export function addWatchedZip(list: readonly string[], input: string): { ok: true; list: string[]; zip: string } | { ok: false; error: 'invalid' | 'duplicate' | 'limit'; zip?: string };
export function removeWatchedZip(list: readonly string[], zip: string): string[];
```
