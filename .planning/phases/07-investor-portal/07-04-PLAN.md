```ts
// apps/web/src/lib/ai/ui/post-check.ts
import type { Finding } from '<Phase 5 finding type module>';
export interface PostCheck {
	id: string;                              // e.g. 'numbers'
	groupLabel: string;                      // 'Number check' — hidden mark prefix becomes `${groupLabel}, must fix: ` / `${groupLabel}, review: `
	passedTitle: string;                     // 'Number check passed'
	passedNote?: string;                     // 'Every number in this memo matches a figure from your analysis.'
	findingsTitle: (n: number) => string;    // n => n === 1 ? 'Number check: 1 to fix' : `Number check: ${n} to fix`
	categoryLabel: string;                   // 'Number not in your analysis'
	explanation: string;                     // popover explanation
	allowRemovePhrase: boolean;              // false for numbers (UI-SPEC: removing a bare number breaks the sentence)
	gatingReason?: (blockCount: number) => string; // 'Fix the {n} numbers marked Must fix to copy, email, or print this memo.'
	doneSentence?: (findings: Finding[]) => string; // 'Every number matches your analysis.' | '{n} numbers to fix.'
	run: (text: string) => Finding[];        // pure, synchronous; findings carry Phase 5's {start,end,severity,category,...}
}
export interface PanelGateState {
	phase: 'empty' | 'streaming' | 'ready';
	text: string;
	artifactId: string | null;
	exitsEnabled: boolean;
	reason: string | null;                   // visible gating reason, null when enabled
	blockCount: number;                      // FH + all post-checks
	unackedWarnCount: number;
}
// GenerationPanel additive props: postChecks?: PostCheck[] (default []), subjectId?: string, artifactMeta?: Record<string, string>, onstatechange?: (s: PanelGateState) => void
// DraftHistory additive prop: subjectId?: string (filters artifacts in memory; no Dexie index change)
// Artifact additive optional fields: subjectId?: string; meta?: Record<string, string>
// PromptDef additive optional field: sample?: (inputs: unknown) => string
// LlmRequest additive optional field: fixtureText?: string (Fixture provider streams it when present)
```
