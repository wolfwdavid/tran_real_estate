export type AutofillTarget = 'rent' | 'value';
export function estimateQueryFromForm(form: AnalyzerForm): EstimateQuery | null;          // null when the address is blank; sends propertyType/bedrooms/bathrooms/squareFeet only when present
export type AutofillOutcome = { kind: 'ok'; estimate: Estimate } | { kind: 'no-address' } | { kind: 'cap'; retry: () => Promise<AutofillOutcome> } | { kind: 'error'; error: unknown };
export async function runAutofill(port: InvestorDataPort, target: AutofillTarget, form: AnalyzerForm): Promise<AutofillOutcome>; // the ONLY spend path; called from onclick
export function applyEstimate(form: AnalyzerForm, sources: DealSources, target: AutofillTarget, e: Estimate): { form: AnalyzerForm; sources: DealSources; kind: 'filled' | 'kept' };
export function useEstimate(form: AnalyzerForm, sources: DealSources, target: AutofillTarget, e: Estimate): { form: AnalyzerForm; sources: DealSources };
export function markEdited(sources: DealSources, key: 'rent' | 'price' | 'address' | 'propertyType' | 'bedrooms' | 'bathrooms' | 'squareFeet'): DealSources; // rentcast/sample/listing -> edited; none stays none
export function priceComparison(priceCents: number, estimateCents: number): string;
