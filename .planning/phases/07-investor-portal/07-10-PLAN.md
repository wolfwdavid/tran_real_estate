```ts
// form-model.ts
export interface ExpenseLineForm { id: string; kind: ExpenseKind; name: string; unit: 'pct' | 'usd'; amount: string }
export interface AnalyzerForm {
	name: string; address: string; propertyType: '' | 'single-family' | 'condo' | 'townhouse' | 'multi-family' | 'manufactured';
	bedrooms: string; bathrooms: string; squareFeet: string;
	price: string; closingAmount: string; closingUnit: 'pct' | 'usd'; repairs: string;
	financing: 'mortgage' | 'cash'; downPct: string; rate: string; termYears: string;
	rent: string; otherIncome: string;
	expenses: ExpenseLineForm[];
}
export function defaultForm(): AnalyzerForm; // closing '3' pct, repairs '0', down '25', rate '', term '30', otherIncome '0', expenses from defaultExpenseLines() (pct amounts '5','8','5','5'; usd amounts '')
export type FieldError = 'not-a-number' | 'negative' | 'down-range' | 'rate-range' | 'term-range' | 'pct-range' | 'bedrooms';
export const FIELD_ERROR_COPY: Record<FieldError, string>;
// 'not-a-number': 'Enter a number, like 250000.' · 'negative': 'Use 0 or more.' · 'down-range': 'Use a percent from 0 to 100.' · 'rate-range': 'Use a rate from 0 to 25%.' · 'term-range': 'Use a term from 1 to 40 years.' · 'pct-range': 'Use a percent from 0 to 100.' · 'bedrooms': 'Use a whole number from 0 to 20.'
export type FieldKey = 'price' | 'rent' | 'closing' | 'repairs' | 'downPct' | 'rate' | 'termYears' | 'otherIncome' | 'bedrooms' | 'bathrooms' | 'squareFeet' | `expense:${string}`;
export const FIELD_NAMES: Record<string, string>; // lowercase names for "Needs {field}": price 'purchase price', rent 'monthly rent', closing 'closing costs', repairs 'repairs before renting', downPct 'down payment', rate 'interest rate', termYears 'loan term', otherIncome 'other monthly income'; expense lines use their own lowercased name
export type ImpactMetric = 'cashFlow' | 'coc' | 'capRate' | 'noi' | 'dscr' | 'grm' | 'onePct' | 'cashNeeded' | 'loan' | 'payment';
export const FIELD_IMPACT: Record<'closing' | 'repairs' | 'downPct' | 'termYears' | 'otherIncome' | 'expense', ImpactMetric[]>;
// closing/repairs -> ['coc','cashNeeded']; downPct/termYears -> ['loan','payment','cashFlow','coc','dscr','cashNeeded']; otherIncome/expense -> ['noi','cashFlow','coc','capRate','dscr']
export interface ParsedForm { ready: boolean; inputs: DealInputs | null; errors: Partial<Record<FieldKey, FieldError>>; masked: Partial<Record<ImpactMetric, FieldKey>> }
export function validateField(form: AnalyzerForm, key: FieldKey): FieldError | null; // used on blur and on save
export function parseForm(form: AnalyzerForm): ParsedForm;
export function inputsToForm(d: { name: string; address: string; propertyType: AnalyzerForm['propertyType'] | null; bedrooms: number | null; bathrooms: number | null; squareFeet: number | null; inputs: DealInputs }): AnalyzerForm;
export function formatOnBlur(kind: 'money' | 'percent' | 'number', raw: string): string; // "250000" -> "250,000"; "$1234.5" -> "1,234.50"; invalid -> raw unchanged
```
