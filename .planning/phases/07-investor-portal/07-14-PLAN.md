export type BuyBoxPropertyType = 'Single Family' | 'Condo' | 'Townhouse' | 'Multi-Family' | 'Manufactured';
export interface BuyBoxCriteria { location: string; minPriceCents: number | null; maxPriceCents: number | null; minBeds: 0 | 1 | 2 | 3 | 4; propertyTypes: BuyBoxPropertyType[] } // minBeds 0 = Any; [] = any type
export interface BuyBoxInput { name: string; criteria: BuyBoxCriteria; targets: BuyBoxTargets; expenseBps: number }
export interface BuyBoxView extends BuyBoxInput { id: string; lastRunAt: number | null; lastSeenAt: number | null; createdAt: number; updatedAt: number }
export type BuyBoxError = 'name-empty' | 'location-empty' | 'location-unrecognized' | 'min-over-max';
export function validateBuyBox(input: BuyBoxInput): BuyBoxError[];
export async function createBuyBox(input: BuyBoxInput): Promise<BuyBoxView>;          // no network; the UI calls checkNow for "Save and check now"
export async function updateBuyBox(id: string, input: BuyBoxInput): Promise<BuyBoxView>; // location change resets lastRunAt to null (new geography)
export async function deleteBuyBox(id: string): Promise<void>;                          // removes saved search, its snapshot and the buyBoxes row
export async function listBuyBoxes(): Promise<BuyBoxView[]>;
export async function checkNow(id: string, opts?: { grant?: unknown }): Promise<{ newCount: number }>; // ONE gateway call through the scheduler's single-run path (intent 'user'); deduped/cached by the gateway
export async function markSeen(id: string): Promise<void>;
export async function newMatchCount(id: string): Promise<number>; // filtered by criteria
// matches.ts
export interface MatchRow { listing: SaleListing; isNew: boolean; quick: QuickMetrics | null; meets: boolean | null }
export function filterListings(listings: readonly SaleListing[], c: BuyBoxCriteria): SaleListing[]; // price range, beds ≥ minBeds, type in list; sorted by listedAt desc (null last)
export function buildMatches(box: BuyBoxInput, listings: readonly SaleListing[], newIds: ReadonlySet<string>, statsByZip: ReadonlyMap<string, MarketStatsInput | null>): { meeting: MatchRow[]; below: MatchRow[]; zipsNeedingStats: string[] };
export async function loadMatches(box: BuyBoxView): Promise<{ meeting: MatchRow[]; below: MatchRow[]; zipsNeedingStats: string[]; truncated: boolean; checked: boolean }>; // peek-only: peekSaleListings + markets peek per distinct zip; 0 calls
