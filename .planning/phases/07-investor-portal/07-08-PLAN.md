export interface FieldSourceFact { source: 'rentcast' | 'sample' | 'listing' | 'you'; asOf?: number; rangeLowCents?: number; rangeHighCents?: number; comps?: number }
export interface MemoFactsInput {
	dealName: string; address: string;
	propertyType?: string | null; bedrooms?: number | null; bathrooms?: number | null; squareFeet?: number | null;
	inputs: DealInputs; analysis: DealAnalysis;
	sources?: { rent?: FieldSourceFact; price?: FieldSourceFact };
	notes?: string;                         // ≤ 1,000 chars, enforced by the UI
}
export interface MemoFacts { factsText: string; lines: { label: string; value: string }[]; notes: string; allowed: AllowedNumber[] }
export function buildMemoFacts(input: MemoFactsInput): MemoFacts;
// memo-sample.ts
export function sampleMemo(facts: MemoFacts): string;
// analysis-hash.ts
export function canonicalJson(value: unknown): string;   // sorted keys, no whitespace
export function analysisHash(inputs: DealInputs): string; // 8 lowercase hex chars, includes ANALYZER_VERSION
// memo-language.ts
export interface AdviceFinding { start: number; end: number; text: string; severity: 'block' | 'warn'; category: 'advice-language'; phrase: string }
export function checkAdviceLanguage(text: string): AdviceFinding[];
