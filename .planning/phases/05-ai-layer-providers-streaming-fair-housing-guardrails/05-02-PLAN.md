```ts
// types.ts
export type Category = 'race' | 'color' | 'religion' | 'sex' | 'familial-status' | 'national-origin' | 'disability'
  | 'source-of-income' | 'age' | 'marital-status' | 'sogi';
export type Severity = 'block' | 'warn';
export interface LexiconEntry {
  id: string;              // stable, e.g. 'familial.no-kids'
  pattern: string;         // RegExp source authored with p``; matcher adds boundaries + 'giu'
  category: Category;
  scope: 'federal' | 'state';
  severity: Severity;
  explanation: string;     // plain language, our own words
  alternative: string;     // '' means "Remove phrase"
  source: string;          // e.g. 'HUD Achtenberg memo (1995), familial status'
  examples: string[];      // >= 1; each MUST be matched by this entry and by no other entry
}
export interface FindingMember {
  entryId: string;
  checker: string;         // 'fair-housing' for lexicon hits; other checkers (Phase 7 postChecks) use their own id
  category: string;        // a Category for fair-housing; free-form for other checkers
  categoryLabel: string;   // user-facing label, e.g. 'Familial status'
  scope: 'federal' | 'state' | 'other';
  severity: Severity;
  explanation: string;
  alternative: string;
  source: string;
  phrase: string;          // text.slice(start, end)
  start: number;
  end: number;
}
export interface Finding {   // one merged group = one <mark> in the UI
  id: string;              // `${start}-${end}`
  start: number;
  end: number;
  severity: Severity;      // highest among members ('block' > 'warn')
  phrase: string;
  members: FindingMember[];// in start order, ties: block first, then lexicon order
}
export const CATEGORY_LABELS: Record<Category, string>;
export const STATE_SCOPE_NOTE = 'Protected under many state and local laws.';
// pattern.ts
export function p(strings: TemplateStringsArray, ...values: string[]): string;
export function assertSafePattern(source: string): void;   // throws Error('Unsafe lexicon pattern: ...')
export function compilePattern(source: string): RegExp;    // boundaries + 'giu'
// check.ts
export function checkFairHousing(text: string, lexicon?: readonly LexiconEntry[], allow?: readonly string[]): Finding[];
export function mergeFindings(members: readonly FindingMember[], text: string): Finding[];
// apply.ts
export function applyAlternative(text: string, finding: Finding): { text: string; cursor: number };
```
