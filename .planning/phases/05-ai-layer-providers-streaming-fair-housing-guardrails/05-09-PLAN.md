// prompts/types.ts
export type PromptBuild = { system: string; user: string; known: KnownPii[] };
export interface PromptDef<I = unknown> {
  id: string;                 // 'shared/demo'
  version: number;            // 1
  feature: string;            // 'demo'
  housingCopy: boolean;       // true → guardrail prepended
  containsClientData: boolean;// drives the Gemini dialog and the redaction list
  maxTokens: number;          // >= 2048 (thinking tokens count against it: RESEARCH Pitfall 2)
  build(inputs: I): PromptBuild;
}
export type BuiltPrompt = {
  id: string; version: number; promptId: string /* `${id}@${version}` */; feature: string;
  system: string; user: string; redactions: Redaction[];
  containsClientData: boolean; maxTokens: number; inputHash: string /* sha256 hex of system + '\n\n' + user (post-redaction) */;
};
// prompts/registry.ts
export function getPrompt(id: string): PromptDef;          // throws Error('Unknown prompt: ' + id)
export async function buildPrompt(id: string, inputs: unknown): Promise<BuiltPrompt>;
export function registerPrompt(def: PromptDef): void;       // portal phases register their own prompts
// prompts/hash.ts
export async function sha256Hex(text: string): Promise<string>;
// runner.ts
export type RunStatus = 'done' | 'stopped' | 'error';
export type GenerationResult = {
  text: string; usage?: LlmUsage; status: RunStatus; finish: LlmFinish; error?: LlmError;
  provider: AiProviderId; model: string; promptId: string; estimatedCostUsd?: number;
};
export type RunUpdate = { type: 'text'; text: string } | { type: 'first-token' } | { type: 'done'; result: GenerationResult };
export function runGeneration(o: { prompt: BuiltPrompt; provider: LlmProvider; model: string; signal: AbortSignal;
  clientDataAck?: boolean; now?: () => number; flushMs?: number; at?: () => Date }): AsyncGenerator<RunUpdate>;
// providers/fixture.ts
export function fixtureProvider(deps?: { sleep?: (ms: number, signal: AbortSignal) => Promise<void>; random?: () => number;
  outputs?: Readonly<Record<string, string>> }): LlmProvider;   // id 'fixture'
// providers/fixture-outputs.ts
export const FIXTURE_OUTPUTS: Readonly<Record<string, string>>;  // keyed by promptId, plus 'default'
