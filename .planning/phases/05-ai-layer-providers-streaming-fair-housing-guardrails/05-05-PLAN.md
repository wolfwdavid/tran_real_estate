export async function runTestConnection(probe, key, model, deps = {}) {
  const f = resolveFetch(deps);
  let check: ProbeResponse;
  try { check = await probe.keyCheck(key, f); } catch { return { outcome: 'network' }; }
  if (check.status !== 200) {
    const kind = classifyHttp(probe.id, check.status, check.body).kind;
    return { outcome: kind === 'auth' ? 'auth' : kind === 'quota' ? 'quota' : 'network' };
  }
  let modelListed: boolean | undefined;
  try { modelListed = await probe.modelListed(key, model, f, check); } catch { modelListed = undefined; }
  try { await probe.canary(f); } catch { return { outcome: 'cors-blocked' }; }
  return { outcome: 'ok', modelListed };
}
export async function disambiguateFetchFailure(probe, key, deps = {}) {
  const online = deps.isOnline ?? (() => globalThis.navigator?.onLine !== false);
  try {
    const check = await probe.keyCheck(key, resolveFetch(deps));
    return check.status === 200 ? 'cors-blocked' : classifyHttp(probe.id, check.status, check.body).kind;
  } catch { return online() ? 'cors-blocked' : 'network'; }
}
