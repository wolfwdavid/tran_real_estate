```ts
export interface BedroomRent { beds: number; median: number | null; listings: number | null }
export interface SnapshotMetrics {
  zip: string; lastUpdated: string | null;
  medianRent: number | null; medianPrice: number | null; priceToRent: number | null; // 1 decimal
  rentChangePct: number | null; rentChangeFrom: string | null; rentChangeTo: string | null; months: number; // 'YYYY-MM'
  saleDom: number | null; rentalDom: number | null;
  saleTotal: number | null; saleNew: number | null; rentalTotal: number | null; rentalNew: number | null;
  rentByBedroom: BedroomRent[]; // ascending beds; beds >= 4 collapsed into one row beds: 4 ("4+ bedrooms")
}
export function buildSnapshotMetrics(m: MarketStats): SnapshotMetrics;
export interface BedroomAggregate { beds: number; count: number; min: number; max: number; median: number; medianDom: number | null }
export interface AggregatedSample { kind: 'sale' | 'rental'; seenAt: string; total: number; byBedroom: BedroomAggregate[] }
export const AGGREGATED_SAMPLE_KEYS = ['kind', 'seenAt', 'total', 'byBedroom'] as const;
export const BEDROOM_AGGREGATE_KEYS = ['beds', 'count', 'min', 'max', 'median', 'medianDom'] as const;
export function aggregateListings(listings: readonly Listing[], kind: 'sale' | 'rental', seenAt: string): AggregatedSample;
export function bedroomLabel(beds: number): string; // 0 'Studio', 1 '1 bedroom', 2-3 '{n} bedrooms', >=4 '4+ bedrooms'
// snapshot-facts.ts
export function formatSignedPct(v: number): string; // 3.2 -> '+3.2%', -1.4 -> '−1.4%', 0 -> '0.0%'
export interface SnapshotFigureValues { medianRent: string; medianPrice: string; priceToRent: string; rentChange: string; saleDom: string; rentalDom: string; saleTotal: string; saleNew: string; rentalTotal: string; rentalNew: string } // '—' (NOT_AVAILABLE) for null
export function snapshotFigureValues(m: SnapshotMetrics): SnapshotFigureValues;
export interface FactLine { label: string; value: string }
export function snapshotFactLines(m: SnapshotMetrics, sample: AggregatedSample | null, asOf: string): FactLine[]; // null figures omitted
export function factsForPrompt(m: SnapshotMetrics, sample: AggregatedSample | null, asOf: string): string; // lines `${label}: ${value}`
export function snapshotAllowedNumbers(m: SnapshotMetrics, sample: AggregatedSample | null, asOf: string): number[];
```
