```ts
// post-checks.ts
export const NUMBER_CATEGORIES = {
  listing: { category: 'Number not in your facts', explanation: "This number isn't in the facts you entered. Change it to match your facts, or remove it." },
  social: { category: 'Number not in your brief', explanation: "This number isn't in your brief. Change it to match your brief, or remove it." },
  reel: { category: "Number you didn't provide", explanation: "This number isn't in what you typed. Scripts may only use results you supply. Change it or remove it." },
  snapshot: { category: 'Number not in this report', explanation: "This number isn't one of the figures in this report. Notes may only use numbers Tran calculated. Change it to a figure from the report, or remove the sentence." }
} as const;
export function numberPostCheck(kind: keyof typeof NUMBER_CATEGORIES, allowed: readonly number[]): PostCheck;  // severity 'block'
export function numberWordsPostCheck(): PostCheck; // severity 'warn', category 'Number written as a word', explanation 'Write numbers as digits so Tran can check them against what you entered.'
export function unsupportedFeaturePostCheck(facts: ListingFacts): PostCheck; // severity 'warn', category 'Not in your facts', explanation `This mentions "${feature}", which isn't in your facts. Remove it, or add it to your facts if the home has it.`
export function listingAllowedNumbers(f: ListingFacts): number[]; // allowedSet([bedrooms, bathrooms, squareFeet, yearBuilt, garageSpaces, price], listingFreeText(f))
export function briefAllowedNumbers(brief: string, link?: string): number[];
export function reelAllowedNumbers(beats: Record<string, string>): number[];
```
