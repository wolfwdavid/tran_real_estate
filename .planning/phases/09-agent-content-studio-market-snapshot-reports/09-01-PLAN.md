```ts
export type Platform = 'instagram' | 'facebook' | 'linkedin';
export const PLATFORMS: readonly Platform[] = ['instagram', 'facebook', 'linkedin']; // fixed card order
export const PLATFORM_LIMITS = { instagram: 2200, facebook: 63206, linkedin: 3000 } as const;
export const HASHTAG_CAPS = { instagram: 5, facebook: 2, linkedin: 3 } as const;
export const PLATFORM_LABELS = { instagram: 'Instagram', facebook: 'Facebook', linkedin: 'LinkedIn' } as const;
export const SAFETY_MARGIN = 40;
export interface DisclosureProfile {
  fullName: string; brokerage: string; licenseType: string; licenseNumber: string; licenseState: string; // 2-letter USPS
  brokeragePhone?: string; teamName?: string; phone?: string; email?: string; website?: string;
}
export type DisclosureVariant = 'ad-inline' | 'ad-block' | 'print';
export type DisclosureField = 'fullName' | 'brokerage' | 'licenseType' | 'licenseNumber' | 'licenseState' | 'brokeragePhone';
export const DISCLOSURE_FIELD_LABELS: Record<DisclosureField, string>;
export const BROKERAGE_PHONE_STATES: readonly string[]; // ['NY', 'NJ']
export function missingFor(p: DisclosureProfile, variant: DisclosureVariant): DisclosureField[];
export function brokerageLine(p: DisclosureProfile): string;
export interface PrintDisclosure { brokerageLine: string; agentLine: string; licenseLine: string; contactLines: string[] }
export interface FormattedDisclosure { variant: DisclosureVariant; text: string; lines: string[]; missing: DisclosureField[]; print?: PrintDisclosure }
export function formatDisclosure(p: DisclosureProfile, variant: DisclosureVariant): FormattedDisclosure;
export function appendDisclosure(draft: string, disclosure: string): string;
export function hashtagBlock(tags: readonly string[]): string;
export function postBudget(platform: Platform, disclosure: string, tags: readonly string[]): number;
export interface AssembledPost { text: string; length: number; limit: number; over: number; keptTags: string[]; droppedTags: number }
export function assemblePost(platform: Platform, body: string, tags: readonly string[], disclosure: string): AssembledPost;
```
