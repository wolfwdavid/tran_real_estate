```ts
// social-sections.ts
export const SOCIAL_MARKERS = { instagram: 'INSTAGRAM', instagramHashtags: 'INSTAGRAM HASHTAGS', facebook: 'FACEBOOK', linkedin: 'LINKEDIN' } as const;
export type SocialSection = { ok: true; body: string; tags: string[] } | { ok: false; reason: 'missing' | 'empty' };
export interface SocialPack { instagram: SocialSection; facebook: SocialSection; linkedin: SocialSection; active: 'instagram' | 'facebook' | 'linkedin' | null }
export function parseSocialPack(text: string, opts?: { streaming?: boolean }): SocialPack;
export function splitTrailingHashtags(body: string): { body: string; tags: string[] };
export function parseHashtagLine(line: string): string[]; // '#A #B, #C' -> ['#A','#B','#C']
export function buildHashtagCandidates(input: { cityOrArea: string; propertyType?: string; postType: 'just-listed' | 'open-house' | 'price-change' | 'market-update' | 'other' }): string[];
// reel.ts
export type ReelLength = 30 | 60;
export type BeatKey = 'hook' | 'pain' | 'before_after' | 'results' | 'mission' | 'cta';
export const BEAT_KEYS: readonly BeatKey[];
export const BEAT_OUTPUT_LABELS: Record<BeatKey, string>; // HOOK, MANUAL PAIN, BEFORE AND AFTER, RESULTS, MISSION, CALL TO ACTION
export const BEAT_NAMES: Record<BeatKey, string>;         // Hook, Manual pain, Before and after, Results, Mission, Call to action
export const REEL_TEMPLATES: Record<ReelLength, readonly (readonly [BeatKey, number, number])[]>;
export const WORDS_PER_SECOND = 2.5; export const CAPTION_LINE_MAX = 42;
export const RESULTS_PLACEHOLDER = '[Add a real result you can document]';
export function formatTime(s: number): string;              // 3 -> '0:03', 60 -> '1:00'
export function formatSlot(start: number, end: number): string; // '0:00 to 0:03'
export function estimateSpokenSeconds(vo: string): number;
export interface ParsedBeat { key: BeatKey; voiceover: string; caption: string; broll: string; start: number; end: number } // start/end = offsets of the beat block in the source text
export type ReelParse = { ok: true; beats: ParsedBeat[] } | { ok: false; reason: 'missing-beats' | 'out-of-order'; missing: BeatKey[] };
export function parseReel(text: string): ReelParse;
export interface TimedBeat { key: BeatKey; name: string; startSec: number; endSec: number; slot: string; voiceover: string; caption: string; broll: string; overrunSeconds: number | null; longCaptionLines: string[]; isPlaceholder: boolean }
export function timeBeats(beats: readonly ParsedBeat[], length: ReelLength): TimedBeat[];
export interface ResultsViolation { start: number; end: number; reason: 'placeholder-missing' | 'digits' }
export function checkResultsBeat(text: string, resultsEmpty: boolean): ResultsViolation[];
export function findAttributedQuotes(text: string): { start: number; end: number }[];
export function formatReelScript(timed: readonly TimedBeat[], length: ReelLength, footer: string): string;
```

  <behavior>
    Social:
    - Well-formed text with all four markers → three ok sections; IG tags from the HASHTAGS section; FB/LI bodies with a trailing hashtag-only line → tags split off via splitTrailingHashtags and body trimmed
    - Marker case and spacing variants ('=== instagram ===', '===Facebook===', '=== LinkedIn ===') are accepted
    - Prose before the first marker ('Here are your posts:') is ignored
    - Missing FACEBOOK marker → facebook { ok: false, reason: 'missing' }, the other two ok; marker present with only whitespace → reason 'empty'
    - Streaming: parseSocialPack('=== INSTAGRAM ===\nHi', { streaming: true }).active === 'instagram'; after '=== FACEBOOK ===\nHey' arrives active === 'facebook'; with streaming false, active === null
    - parseHashtagLine('#JustListed #HarlowSprings, #Condo #') → ['#JustListed', '#HarlowSprings', '#Condo'] (bare '#' dropped)
    - buildHashtagCandidates({ cityOrArea: 'Harlow Springs', propertyType: 'Condo', postType: 'open-house' }) → ['#OpenHouse', '#HarlowSpringsRealEstate', '#HarlowSprings', '#Condo', '#RealEstate'] (order fixed; non-alphanumerics stripped; 'Single family' → '#SingleFamilyHome'; 'Multi-family' → '#MultiFamily'; empty city → no city tags; post types map just-listed '#JustListed', price-change '#PriceImprovement', market-update '#MarketUpdate', other none)
    Reel:
    - REEL_TEMPLATES[30] = hook 0-3, pain 3-8, before_after 8-15, results 15-21, mission 21-26, cta 26-30; REEL_TEMPLATES[60] = hook 0-3, pain 3-12, before_after 12-25, results 25-40, mission 40-52, cta 52-60; test: contiguous, first starts 0, hook ends 3, last ends at length, order === BEAT_KEYS
    - parseReel on six blocks (labels case-insensitive, e.g. 'Manual Pain', each with Voiceover:/Caption:/B-roll: lines in any order) → ok with six beats in order; a missing RESULTS block → { ok:false, reason:'missing-beats', missing:['results'] }; blocks out of order → 'out-of-order'
    - timeBeats(parsed, 30): slots '0:00 to 0:03' … '0:26 to 0:30'; a 20-word hook voiceover (8s at 2.5 w/s vs 3s slot) → overrunSeconds 5; within-slot → null; a caption line of 43 chars → listed in longCaptionLines; text is never shortened
    - estimateSpokenSeconds('one two three four five') === 2
    - checkResultsBeat(text, true) where RESULTS voiceover/caption contain RESULTS_PLACEHOLDER and no digits → []; results beat containing '47 families' → violation 'digits' with offsets over '47'; results beat without the placeholder → 'placeholder-missing'; checkResultsBeat(anything, false) → [] (the number check covers typed results)
    - findAttributedQuotes: '"Best agent ever," said Maria.' → one range; 'My clients say I answer fast.' → one range; '“Call me today.”' with no attribution → []
    - formatReelScript(timed30, 30, 'FOOTER') starts with 'REEL SCRIPT (30 seconds)\n', has per beat '{slot} {BEAT_OUTPUT_LABEL}\nVoiceover: …\nCaption: …\nB-roll: …', then a blank line, 'B-ROLL SHOT LIST', numbered lines '1. 0:00 {broll}', then a blank line and 'FOOTER' last (endsWith('FOOTER'))
