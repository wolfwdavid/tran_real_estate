New additive panel prop (default undefined = current behavior, i.e. default mode with exactly three exits):
```ts
export interface SectionSpec {
  key: 'instagram' | 'facebook' | 'linkedin'; label: string; constraintLine: string;
  emailFooter: { text: string; label: string }; // same shape as the panel's emailFooter prop: text = ad-inline disclosure, label = 'Added to the end of this post'
  extract: (fullText: string, streaming: boolean) => { state: 'empty' | 'streaming' | 'ok' | 'error'; text: string }; // body + '\n\n' + tag line
  assemble: (sectionText: string) => { text: string; length: number; limit: number; over: number; keptTags: string[]; droppedTags: number; tagsOverCap: number };
  postChecks?: PostCheck[];
}
sections?: SectionSpec[];
```
