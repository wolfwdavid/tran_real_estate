}
export type VaultEvent = { kind: 'keys-changed' | 'mode-changed' | 'forgotten' | 'locked' | 'unlocked' | 'lock-changed'; provider?: ProviderId; remote: boolean };
export type VaultErrorCode = 'empty-key' | 'invalid-key' | 'vault-locked' | 'storage-blocked' | 'wrong-passphrase' | 'cooldown' | 'no-lock' | 'corrupt';
export class VaultError extends Error { constructor(readonly code: VaultErrorCode) { super(code); this.name = 'VaultError'; } }
export interface VaultDeps {
	session: Pick<Storage, 'getItem' | 'setItem' | 'removeItem'>;
	openDb: () => SecretsDb;
	channel: ChannelLike | null;

}
export const SESSION_KEY = 'tran.vault.session.v1';
export class Vault {
	constructor(deps: VaultDeps);
	init(): Promise<void>;
	getStatus(): VaultStatus;                                  // returns a fresh snapshot object
	getKey(p: ProviderId): Promise<string | undefined>;        // undefined when absent OR locked
