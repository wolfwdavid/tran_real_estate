```ts
import Dexie, { type Table } from 'dexie';
import type { Syncable } from '@tran/core';
export interface MetaRow { key: string; value: unknown }               // LOCAL-ONLY bookkeeping; never exported
export interface PrefRecord extends Syncable { key: string; value: unknown }
export type DbOptions = { indexedDB?: IDBFactory; IDBKeyRange?: typeof IDBKeyRange };
export class TranDb extends Dexie {
	meta!: Table<MetaRow, string>;
	prefs!: Table<PrefRecord, string>;
	constructor(options?: DbOptions, name = 'tran') {
		super(name, options);
		// Version history is APPEND-ONLY. Later phases add this.version(n + 1).stores({...}) below; never edit an existing block.
		this.version(1).stores({ meta: 'key', prefs: 'id, key, updatedAt' });
	}
}
export class StorageBlockedError extends Error { constructor() { super('storage-blocked'); this.name = 'StorageBlockedError'; } }
export function getDb(options?: DbOptions): TranDb;           // lazy singleton; browser callers only (onMount / event handlers)
export async function openDb(db = getDb()): Promise<TranDb>; // maps SecurityError | InvalidStateError | UnknownError | MissingAPIError → StorageBlockedError
export function _resetDb(): void;                            // tests only
```
