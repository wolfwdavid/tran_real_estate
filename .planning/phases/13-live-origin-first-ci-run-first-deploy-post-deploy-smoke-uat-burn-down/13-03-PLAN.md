```yaml
name: live-smoke

# Read-only smoke of the deployed site (SMOKE-04). Runs after every completed ci run; the job is
# skipped unless Pages is on and the ci run was a green push to main (the only runs that deploy).
# It checks out and expects the deployed commit, never the default-branch tip.
on:
  workflow_run:
    workflows: [ci]
    types: [completed]
    branches: [main]
  workflow_dispatch:
    inputs:
      base_url:
        description: Deployed origin plus base path, with a trailing slash
        required: true
        type: string
      expect_version:
        description: 40-hex build SHA to wait for (default is the dispatched commit)
        required: false
        type: string

permissions:
  contents: read

concurrency:
  group: live-smoke
  cancel-in-progress: true

jobs:
  smoke:
    name: read-only smoke of the deployed site
    if: vars.PAGES_ENABLED == 'true' && (github.event_name == 'workflow_dispatch' || (github.event.workflow_run.conclusion == 'success' && github.event.workflow_run.event == 'push' && github.event.workflow_run.head_branch == 'main'))
    runs-on: ubuntu-24.04
    timeout-minutes: 25
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.event.workflow_run.head_sha || github.sha }}
          persist-credentials: false
      - uses: pnpm/action-setup@ea17c68df8912ef543352723c149a84f56e3d413 # v6.1.0
      - uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020 # v7.0.0
        with:
          node-version-file: .nvmrc
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - run: pnpm --filter web exec playwright install --with-deps chromium
      # Remote mode: waits up to 10 minutes for <base>_app/version.json to report the deployed SHA,
      # then runs the read-only suite with zero retries. Exit 2 = env precondition, 3 = not propagated.
      - name: Smoke the deployed build
        run: pnpm smoke
        env:
          SMOKE_BASE_URL: ${{ inputs.base_url || format('{0}{1}/', vars.PUBLIC_SITE_URL, vars.BASE_PATH) }}
          SMOKE_SITE_URL: ${{ inputs.base_url || format('{0}{1}/', vars.PUBLIC_SITE_URL, vars.BASE_PATH) }}
          SMOKE_EXPECT_VERSION: ${{ github.event.workflow_run.head_sha || inputs.expect_version || github.sha }}
          SMOKE_ANALYTICS_HOST: ${{ vars.PUBLIC_ANALYTICS_HOST }}
      - name: Upload traces on failure
        if: failure()
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: live-smoke-traces
          path: apps/web/test-results/
          retention-days: 14
          if-no-files-found: ignore
```
Note on D-09: the `PAGES_ENABLED` clause also gates `workflow_dispatch` (stricter reading of D-09: no red runs while Pages is off). Note the header comment must not contain `https://` (test 6).
Run the gate test (green), `pnpm lint:workflows`, and `pnpm vitest run --project scripts scripts/workflow-gate.test.ts scripts/extension-release-gate.test.ts` (unchanged and green).
Commit both files together: `ci: live-smoke workflow runs the read-only smoke after every deploy`.
